cmake_minimum_required(VERSION 3.28)

project(ScreenTo3DTensorRt11Smoke LANGUAGES CXX)

if(DEFINED STEREOTO3D_TENSORRT_ROOT AND NOT DEFINED SCREENTO3D_TENSORRT_ROOT)
    set(SCREENTO3D_TENSORRT_ROOT "${STEREOTO3D_TENSORRT_ROOT}" CACHE PATH "TensorRT SDK root")
endif()
if(DEFINED STEREOTO3D_CUDA_ROOT AND NOT DEFINED SCREENTO3D_CUDA_ROOT)
    set(SCREENTO3D_CUDA_ROOT "${STEREOTO3D_CUDA_ROOT}" CACHE PATH "CUDA Toolkit root")
endif()
if(DEFINED DEPTHSBS_TENSORRT_ROOT AND NOT DEFINED SCREENTO3D_TENSORRT_ROOT)
    set(SCREENTO3D_TENSORRT_ROOT "${DEPTHSBS_TENSORRT_ROOT}" CACHE PATH "TensorRT SDK root")
endif()
if(DEFINED DEPTHSBS_CUDA_ROOT AND NOT DEFINED SCREENTO3D_CUDA_ROOT)
    set(SCREENTO3D_CUDA_ROOT "${DEPTHSBS_CUDA_ROOT}" CACHE PATH "CUDA Toolkit root")
endif()
set(SCREENTO3D_TENSORRT_ROOT "" CACHE PATH "TensorRT SDK root")
set(SCREENTO3D_CUDA_ROOT "" CACHE PATH "CUDA Toolkit root")
if(NOT SCREENTO3D_CUDA_ROOT AND DEFINED ENV{CUDA_PATH})
    set(SCREENTO3D_CUDA_ROOT "$ENV{CUDA_PATH}")
endif()
if(NOT SCREENTO3D_TENSORRT_ROOT OR NOT SCREENTO3D_CUDA_ROOT)
    message(FATAL_ERROR "Set SCREENTO3D_TENSORRT_ROOT and SCREENTO3D_CUDA_ROOT")
endif()

find_path(TENSORRT_INCLUDE_DIR NvInfer.h
    PATHS "${SCREENTO3D_TENSORRT_ROOT}/include" NO_DEFAULT_PATH REQUIRED)
find_library(TENSORRT_LIBRARY nvinfer_11
    PATHS "${SCREENTO3D_TENSORRT_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
find_library(TENSORRT_ONNX_LIBRARY nvonnxparser_11
    PATHS "${SCREENTO3D_TENSORRT_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
find_path(CUDA_INCLUDE_DIR cuda_runtime_api.h
    PATHS "${SCREENTO3D_CUDA_ROOT}/include" NO_DEFAULT_PATH REQUIRED)

add_executable(screento3d_tensorrt11_smoke main.cpp)
target_compile_features(screento3d_tensorrt11_smoke PRIVATE cxx_std_20)
target_include_directories(screento3d_tensorrt11_smoke PRIVATE
    "${TENSORRT_INCLUDE_DIR}"
    "${CUDA_INCLUDE_DIR}"
)
target_link_libraries(screento3d_tensorrt11_smoke PRIVATE
    "${TENSORRT_LIBRARY}"
    "${TENSORRT_ONNX_LIBRARY}"
)

if(MSVC)
    target_compile_options(screento3d_tensorrt11_smoke PRIVATE
        /W4 /WX /permissive- /EHsc /utf-8)
else()
    target_compile_options(screento3d_tensorrt11_smoke PRIVATE
        -Wall -Wextra -Wpedantic -Werror)
endif()
